cmake_minimum_required(VERSION 3.21)
project(Mengzhi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets PrintSupport Test Concurrent)
if(APPLE AND TARGET WrapOpenGL::WrapOpenGL)
  # Qt 6.8's imported OpenGL target adds the legacy AGL linker flag even on SDKs
  # that no longer expose it. Our raster Widgets code uses no AGL symbols.
  get_target_property(_qt_gl_links WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
  list(FILTER _qt_gl_links EXCLUDE REGEX "AGL")
  set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${_qt_gl_links}")
endif()
qt_add_executable(Mengzhi MACOSX_BUNDLE WIN32
  native/branding.qrc native/src/main.cpp native/src/model.h native/src/canvas.h native/src/canvas.cpp
  native/src/window.h native/src/window.cpp native/src/overlay.h native/src/overlay.cpp
  native/src/export.h native/src/export.cpp native/src/hotkey.h native/src/hotkey.cpp
  native/src/ocr_layout.cpp native/src/ocr_review.h native/src/capture_targets.cpp native/src/capture_options.cpp native/src/pin_editor.cpp
  native/src/ui_study.cpp native/src/ui_study_pearl.cpp native/src/ui_study_graphite.cpp native/src/ui_study_folio.cpp native/src/studio_interactions.cpp native/src/capture_toolbar.cpp)
target_link_libraries(Mengzhi PRIVATE Qt6::Widgets Qt6::PrintSupport Qt6::Concurrent)
set_target_properties(Mengzhi PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "萌知" MACOSX_BUNDLE_GUI_IDENTIFIER "app.mengzhi.capture" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/native/Info.plist.in")
if(APPLE)
  enable_language(OBJCXX)
  target_sources(Mengzhi PRIVATE native/src/material_mac.mm native/src/ocr_mac.mm)
  set_source_files_properties(native/src/ocr_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(Mengzhi PRIVATE "-framework Vision")
  set_source_files_properties(native/src/material_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(Mengzhi PRIVATE "-framework Carbon" "-framework ApplicationServices")
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shizi-ocr COMMAND swiftc -O -module-cache-path ${CMAKE_CURRENT_BINARY_DIR}/swift-cache ${CMAKE_CURRENT_SOURCE_DIR}/native/helpers/ocr.swift -o ${CMAKE_CURRENT_BINARY_DIR}/shizi-ocr DEPENDS native/helpers/ocr.swift)
  add_custom_target(ocr_helper DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/shizi-ocr)
  add_dependencies(Mengzhi ocr_helper)
  add_custom_command(TARGET Mengzhi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_BINARY_DIR}/shizi-ocr "$<TARGET_BUNDLE_DIR:Mengzhi>/Contents/MacOS/shizi-ocr")
else()
  target_sources(Mengzhi PRIVATE native/src/material.cpp native/src/ocr.cpp)
  if(WIN32)
    target_link_libraries(Mengzhi PRIVATE dwmapi)
  endif()
  add_custom_command(TARGET Mengzhi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/native/helpers/windows.ps1 $<TARGET_FILE_DIR:Mengzhi>/windows.ps1)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(export_tests native/tests/export_test.cpp native/src/export.cpp native/src/model.h)
  target_link_libraries(export_tests PRIVATE Qt6::Test Qt6::Widgets Qt6::PrintSupport)
  add_test(NAME export_tests COMMAND export_tests -platform offscreen)
  if(WIN32)
    qt_add_executable(ocr_tests native/tests/ocr_test.cpp native/src/ocr.cpp native/src/ocr_layout.cpp)
    target_link_libraries(ocr_tests PRIVATE Qt6::Test Qt6::Widgets)
    add_dependencies(ocr_tests Mengzhi)
    add_test(NAME ocr_tests COMMAND ocr_tests -platform offscreen)
    set_tests_properties(ocr_tests PROPERTIES TIMEOUT 120)
    qt_add_executable(capture_tests native/tests/capture_test.cpp native/src/overlay.cpp native/src/canvas.cpp native/src/capture_targets.cpp native/src/pin_editor.cpp native/src/capture_toolbar.cpp)
    target_link_libraries(capture_tests PRIVATE Qt6::Test Qt6::Widgets dwmapi)
    add_test(NAME capture_tests COMMAND capture_tests -platform offscreen)
    qt_add_executable(canvas_interaction_tests native/tests/canvas_interaction_test.cpp native/src/canvas.cpp)
    target_link_libraries(canvas_interaction_tests PRIVATE Qt6::Test Qt6::Widgets)
    add_test(NAME canvas_interaction_tests COMMAND canvas_interaction_tests -platform offscreen)
  endif()
endif()
install(TARGETS Mengzhi BUNDLE DESTINATION . RUNTIME DESTINATION .)

if(APPLE)
 target_sources(Mengzhi PRIVATE assets/branding/mengzhi.icns)
 set_source_files_properties(assets/branding/mengzhi.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
elseif(WIN32)
 configure_file(native/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
 target_sources(Mengzhi PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
endif()
